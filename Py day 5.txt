#Q2


products = [
    ("Sensor", 4, 150.0),
    ("Cable", 12, 25.0),
    ("Board", 3, 800.0)
]
total = 0
for item in products:
    name = item[0]
    quantity = item[1]
    price = item[2]
    value = quantity * price
    total = total + value
    print(name, "Inventory Value:", value)
    if quantity < 5:
        print(name, "has fewer than 5 units in stock")
print("Total Inventory Value:", total)

#Q9


a = [
    [1, 2, 3],
    [4, 5, 6]
]
b = [
    [7, 8],
    [9, 10],
    [11, 12]
]
if len(a[0]) != len(b):
    print("Matrix multiplication is not possible")
else:
    result = []
    for i in range(len(a)):
        row = []
        for j in range(len(b[0])):
            total = 0
            for k in range(len(b)):
                total += a[i][k] * b[k][j]
            row.append(total)
        result.append(row)
    for row in result:
        print(row)




#W10




matrix = [
    [1, 3, 1],
    [2, 5, 2],
    [4, 2, 1]
]
rows = len(matrix)
cols = len(matrix[0])
dp = [[0] * cols for i in range(rows)]
dp[0][0] = matrix[0][0]
for i in range(1, rows):
    dp[i][0] = dp[i - 1][0] + matrix[i][0]
for j in range(1, cols):
    dp[0][j] = dp[0][j - 1] + matrix[0][j]
for i in range(1, rows):
    for j in range(1, cols):
        dp[i][j] = matrix[i][j] + max(dp[i - 1][j], dp[i][j - 1])
path = []
i = rows - 1
j = cols - 1
while i > 0 or j > 0:
    path.append(matrix[i][j])
    if i == 0:
        j -= 1
    elif j == 0:
        i -= 1
    elif dp[i - 1][j] > dp[i][j - 1]:
        i -= 1
    else:
        j -= 1
path.append(matrix[0][0])
path.reverse()
print("Maximum Sum:", dp[rows - 1][cols - 1])
print("Path:", path)
